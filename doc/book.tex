\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{fancyvrb}
% A fixed measure is a paper problem: html wraps by itself. tex4ht has a hook
% for fancyvrb and none for fvextra, whose line breaking left every example in
% the book empty.
\ifdefined\HCode\else\usepackage{fvextra}\fi
% D identifiers carry underscores, and a bare _ is a math shift in LaTeX. Every
% symbol name in the book would otherwise have to be escaped by hand.
\usepackage{underscore}
\usepackage{xcolor}
\usepackage[hidelinks]{hyperref}

% Mapping=tex-text turns a straight " into a closing quote, which silently
% reverses the opening mark on every line that carries the operator's words.

\definecolor{grule}{HTML}{B4B4B4}

\titleformat{\part}[display]{\Huge\scshape}{}{0pt}{}
\titleformat{\chapter}[display]{\LARGE\scshape}{}{0pt}{}[\vspace{2pt}\color{grule}\hrule]
\titlespacing*{\chapter}{0pt}{0pt}{28pt}

\newcommand{\gsection}[1]{\vspace{6pt}\noindent\textsc{\small #1}\par\nopagebreak\vspace{2pt}}

% A TODO is not reference. It is a note to whoever edits the file, and it reads
% as documentation unless it is marked as something else.
\definecolor{gtodo}{HTML}{8A7A50}
\newcommand{\gtodo}[1]{{\color{gtodo}\small\sffamily #1\par}}
\newcommand{\gverb}[1]{\texttt{#1}}

% No gobble: press writes chapter code unindented, and eating a character took
% the s off every `static assert`. breaklines keeps a long assertion on the page.
\ifdefined\HCode
  \DefineVerbatimEnvironment{gcode}{Verbatim}{fontsize=\small}
\else
  \DefineVerbatimEnvironment{gcode}{Verbatim}{
    fontsize=\small, xleftmargin=14pt, frame=leftline,
    breaklines=true, breakanywhere=true, breaksymbolleft={},
    rulecolor=\color{grule}, framesep=8pt}
\fi

\DefineVerbatimEnvironment{gsig}{Verbatim}{
  fontsize=\small, xleftmargin=0pt, formatcom=\color{gink}}

\newenvironment{gparams}{\begin{description}[leftmargin=!,font=\normalfont\ttfamily]}{\end{description}}

% A glossary term is a word of the book's own language, so it is set as a word
% and not as code.
\newenvironment{gglossary}{\begin{description}[leftmargin=!,font=\normalfont\scshape]}{\end{description}}

% A case is one row: the block on the left, what it is for on the right. A
% minipage cannot be broken, so a page never cuts an example in half.
\newlength{\gpbtw}
\newlength{\gnotew}
\setlength{\gpbtw}{0.58\textwidth}
\setlength{\gnotew}{0.38\textwidth}

% The note sits beside the block it is about, so it is set narrow and ragged.
% Justifying a column this measure opens rivers no reader should have to cross.
\newcommand{\gnote}[1]{\raggedright\small #1\par}


% Which sheet carries which printed number. The reader asks for the number on
% the paper, and only the typesetter that broke the pages knows where it fell.
\usepackage{atbegshi}
\newwrite\folioout
\newcounter{sheet}
\ifdefined\HCode\else
  \immediate\openout\folioout=\jobname.folios
  \AtBeginShipout{\stepcounter{sheet}%
    \immediate\write\folioout{\thesheet\space\thepage}}
\fi
